     1→     1→"""Transaction engine — the SINGLE payment code path for every caller.
     2→     2→
     3→     3→Flow (autonomous, amount <= SPEND_LIMIT_PAISE):
     4→     4→  idempotency pre-check -> MCP create_order (atomic stock lock) -> INVENTORY_LOCK
     5→     5→  audit -> GUARDRAIL_PASS audit -> [ACID txn: OCC debit + tx insert + Razorpay
     6→     6→  TEST order + order update] -> ORDER_GENERATED audit -> return 'pending'
     7→     7→  (webhook payment.captured later completes it).
     8→     8→
     9→     9→Flow (OTP-gated, amount > SPEND_LIMIT_PAISE):
    10→    10→  same until INVENTORY_LOCK, then: transaction doc status 'awaiting_otp' +
    11→    11→  otp_challenges doc (bcrypt hash) -> GUARDRAIL_OTP_REQUIRED audit + ws
    12→    12→  'otp:required' -> 202. THE WALLET IS NOT TOUCHED until a human verifies.
    13→    13→  On verify: OTP_VERIFIED audit -> same ACID block with a server-side
    14→    14→  OtpAuthorization -> ORDER_GENERATED.
    15→    15→
    16→    16→Idempotency / replay-attack prevention (both lines active):
    17→    17→  1. Pre-check: transactions.idempotencyKey lookup -> stored resultSnapshot is
    18→    18→     returned VERBATIM (no re-debit, no re-Razorpay call).
    19→    19→  2. DB level: unique index on idempotencyKey -> E11000 on concurrent same-key
    20→    20→     inserts is caught and the winner's snapshot is returned.
    21→    21→
    22→    22→Concurrency: the debit is wallet_service.execute_debit — atomic OCC
    23→    23→find_one_and_update with 5 jittered-backoff retries, fresh guardrail check per
    24→    24→attempt, and multi-document ACID semantics via the passed Mongo session.
    25→    25→"""
    26→    26→import functools
    27→    27→import re
    28→    28→import uuid
    29→    29→from datetime import datetime, timedelta, timezone
    30→    30→from typing import Any

    45→    45→
    46→    46→IDEMPOTENCY_KEY_RE = re.compile(r"^[A-Za-z0-9._:-]{8,200}$")
    47→    47→SESSION_ID_RE = re.compile(r"^[A-Za-z0-9._:-]{6,128}$")
    48→    48→

    50→    50→class TransactionError(Exception):
    51→    51→    def __init__(self, code: str, message: str, http_status: int = 400, data: Any = None) -> None:
    52→    52→        super().__init__(message)
    53→    53→        self.code = code
    54→    54→        self.http_status = http_status
    55→    55→        self.data = data

    58→    58→def _utcnow() -> datetime:
    59→    59→    return datetime.now(timezone.utc)

    62→    62→# ---------------------------------------------------------------------------
    63→    63→# Reusable idempotency-key enforcement (Flask decorator — the Python twin of
    64→    64→# mcp-server's middleware/idempotency.ts; write endpoints use it, none hand-roll).
    65→    65→# ---------------------------------------------------------------------------
    66→    66→
    67→    67→def require_idempotency_key(view_fn):
    68→    68→    @functools.wraps(view_fn)
    69→    69→    def wrapper(*args, **kwargs):
    70→    70→        body = request.get_json(silent=True) or {}
    71→    71→        raw = request.headers.get("Idempotency-Key") or body.get("idempotencyKey")
    72→    72→        key = raw.strip() if isinstance(raw, str) else None
    73→    73→        if not key or not IDEMPOTENCY_KEY_RE.match(key):
    74→    74→            return (
    75→    75→                jsonify(
    76→    76→                    {
    77→    77→                        "error": "MISSING_IDEMPOTENCY_KEY",
    78→    78→                        "hint": "Send an Idempotency-Key header or idempotencyKey field (8-200 chars, e.g. a UUID).",
    79→    79→                    }
    80→    80→                ),
    81→    81→                400,
    82→    82→            )
    83→    83→        return view_fn(*args, **kwargs, idempotency_key=key)
    84→    84→
    85→    85→    return wrapper

    88→    88→# ---------------------------------------------------------------------------
    89→    89→# Validation helpers

    92→    92→def _validate_items(items: Any) -> list[dict[str, Any]]:
    93→    93→    if not isinstance(items, list) or not items or len(items) > 20:
    94→    94→        raise TransactionError("INVALID_ITEMS", "items must be a non-empty list (max 20).")
    95→    95→    cleaned = []
    96→    96→    for raw in items:
    97→    97→        if not isinstance(raw, dict):
    98→    98→            raise TransactionError("INVALID_ITEMS", "each item must be an object {sku, qty}.")
    99→    99→        sku = raw.get("sku")
   100→   100→        qty = raw.get("qty")
   101→   101→        if not isinstance(sku, str) or not sku.strip():
   102→   102→            raise TransactionError("INVALID_ITEMS", "each item needs a string sku.")
   103→   103→        if not isinstance(qty, int) or isinstance(qty, bool) or qty < 1 or qty > 100:
   104→   104→            raise TransactionError("INVALID_ITEMS", f"qty for {sku} must be an integer 1..100.")
   105→   105→        cleaned.append({"sku": sku.strip(), "qty": qty})
   106→   106→    return cleaned
   107→   107→
   108→   108→
   109→   109→def _validate_session_id(session_id: Any) -> str:
   110→   110→    if not isinstance(session_id, str) or not SESSION_ID_RE.match(session_id):
   111→   111→        raise TransactionError("INVALID_SESSION_ID", "sessionId must match [A-Za-z0-9._:-]{6,128}.")
   112→   112→    return session_id
   113→   113→
   114→   114→
   115→   115→def _response_for(doc: dict) -> dict:
   116→   116→    """Stored resultSnapshot replayed VERBATIM on idempotent hits."""
   117→   117→    snapshot = doc.get("resultSnapshot")
   118→   118→    if isinstance(snapshot, dict):
   119→   119→        return snapshot
   120→   120→    return {"status": doc.get("status"), "transactionId": str(doc["_id"]), "idempotencyKey": doc.get("idempotencyKey")}
   121→   121→

   123→   123→# ---------------------------------------------------------------------------
   124→   124→# Order helpers (orders live in the shared cluster; agent-service transitions
   125→   125→# payment state on them directly — documented in docs/ARCHITECTURE.md)

   128→   128→def _get_order_doc(order_number: str) -> dict | None:
   129→   129→    return db.orders().find_one({"orderNumber": order_number})
   130→   130→
   131→   131→
   132→   132→_CANCELABLE_STATUSES = ("created", "payment_pending", "payment_failed", "recovery_in_progress")

   135→   135→def cancel_order_and_release_stock(order_number: str, reason: str, agent_id: str, session_id: str) -> bool:
   136→   136→    """Conditionally cancel the order and release reserved stock exactly once."""
   137→   137→    doc = db.orders().find_one_and_update(
   138→   138→        {"orderNumber": order_number, "status": {"$in": list(_CANCELABLE_STATUSES)}},
   139→   139→        {"$set": {"status": "cancelled", "updatedAt": _utcnow()}},
   140→   140→        return_document=ReturnDocument.AFTER,
   141→   141→    )
   142→   142→    if doc is None:
   143→   143→        return False  # already cancelled / paid / recovered — never double-release
   144→   144→    for item in doc.get("items", []):
   145→   145→        db.catalog_items().update_one(
   146→   146→            {"sku": item["sku"], "reservedStock": {"$gte": item["qty"]}},
   147→   147→            {"$inc": {"stock": item["qty"], "reservedStock": -item["qty"], "version": 1}},
   148→   148→        )
   149→   149→    log_step(
   150→   150→        session_id,
   151→   151→        agent_id,
   152→   152→        "ORDER_CANCELLED",
   153→   153→        {"orderNumber": order_number, "reason": reason},
   154→   154→        order_id=doc.get("_id"),
   155→   155→    )
   156→   156→    return True

   159→   159→# ---------------------------------------------------------------------------
   160→   160→# Core: execute_transaction — the one entrypoint for money movement
   161→   161→# ---------------------------------------------------------------------------
   162→   162→
   163→   163→def execute_transaction(
   164→   164→    agent_id: str,
   165→   165→    session_id: str,
   166→   166→    items: Any,
   167→   167→    idempotency_key: str,
   168→   168→    source: str = "api",
   169→   169→) -> dict:
   170→   170→    """Execute a guarded purchase. Returns a JSON-safe response dict.
   171→   171→
   172→   172→    `source` ('api' | 'onyx' | 'mcp') is audit metadata ONLY — it never changes
   173→   173→    which rules apply. The guardrail is enforced identically for every source.
   174→   174→    """
   175→   175→    _validate_session_id(session_id)
   176→   176→    cleaned_items = _validate_items(items)
   177→   177→    if not isinstance(agent_id, str) or len(agent_id.strip()) < 2:
   178→   178→        raise TransactionError("INVALID_AGENT_ID", "agentId must be a non-empty string.")
   179→   179→    if not isinstance(idempotency_key, str) or not IDEMPOTENCY_KEY_RE.match(idempotency_key):
   180→   180→        raise TransactionError("MISSING_IDEMPOTENCY_KEY", "idempotencyKey (8-200 chars) is required.")
   181→   181→    agent_id = agent_id.strip()
   182→   182→
   183→   183→    # ---- Idempotency line 1: replay the stored result verbatim. ----
   184→   184→    existing = db.transactions().find_one({"idempotencyKey": idempotency_key})
   185→   185→    if existing is not None:
   186→   186→        return _response_for(existing)
   187→   187→
   188→   188→    # ---- Create the order via the MCP server (atomic stock lock). ----
   189→   189→    try:
   190→   190→        order = mcp_client.create_order(cleaned_items, agent_id, idempotency_key)
   191→   191→    except mcp_client.McpRemoteError as err:
   192→   192→        if err.code == -32001:  # INSUFFICIENT_STOCK — deterministic, store the failure
   193→   193→            sku = (err.data or {}).get("sku") if isinstance(err.data, dict) else None
   194→   194→            return _record_failed_transaction(
   195→   195→                agent_id, session_id, idempotency_key, None, 0, cleaned_items,
   196→   196→                f"INSUFFICIENT_STOCK{'/' + sku if sku else ''}", source,
   197→   197→            )
   198→   198→        # Other remote errors (rate limit, unauthorized) are NOT stored — retrying
   199→   199→        # the same key should be possible once the condition clears.
   200→   200→        raise TransactionError("MCP_" + err.message, str(err), 502, err.data)
   201→   201→    except mcp_client.McpUnavailable as err:
   202→   202→        # Transient transport failure — deliberately not persisted so a retry
   203→   203→        # with the same idempotency key re-attempts instead of replaying failure.
   204→   204→        raise TransactionError("MCP_UNAVAILABLE", str(err), 503)
   205→   205→
   206→   206→    order_number = order.get("orderNumber", "")
   207→   207→    amount_paise = int(order.get("totalPaise", 0))
   208→   208→    order_doc = _get_order_doc(order_number)
   209→   209→    order_oid = order_doc.get("_id") if order_doc else None
   210→   210→
   211→   211→    log_step(
   212→   212→        session_id, agent_id, "INVENTORY_LOCK",
   213→   213→        {
   214→   214→            "orderNumber": order_number,
   215→   215→            "items": order.get("items", []),
   216→   216→            "totalPaise": amount_paise,
   217→   217→            "duplicateOrder": bool(order.get("duplicate")),
   218→   218→            "source": source,
   219→   219→        },
   220→   220→        order_id=order_oid,
   221→   221→    )
   222→   222→
   223→   223→    # ---- Guardrail path selection (UX only; enforcement lives in wallet_service). ----
   224→   224→    if amount_paise > config.SPEND_LIMIT_PAISE:
   225→   225→        return _create_otp_gated_transaction(
   226→   226→            agent_id, session_id, idempotency_key, order_number, order_oid,
   227→   227→            amount_paise, source,
   228→   228→        )
   229→   229→
   230→   230→    return _execute_payment(
   231→   231→        agent_id, session_id, idempotency_key, order_number, order_oid,
   232→   232→        amount_paise, tx_type="autonomous", authorization=None, source=source,
   233→   233→        pre_authorized=False,
   234→   234→    )
   235→   235→
   236→   236→
   237→   237→# ---------------------------------------------------------------------------
   238→   238→# OTP-gated branch: amount above the limit — wallet untouched until verified
   239→   239→# ---------------------------------------------------------------------------
   240→   240→
   241→   241→def _create_otp_gated_transaction(
   242→   242→    agent_id: str,
   243→   243→    session_id: str,
   244→   244→    idempotency_key: str,
   245→   245→    order_number: str,
   246→   246→    order_oid: ObjectId | None,
   247→   247→    amount_paise: int,
   248→   248→    source: str,
   249→   249→) -> dict:
   250→   250→    tx_id = ObjectId()
   251→   251→    wallet = wallet_service.get_wallet(agent_id)
   252→   252→    version_before = wallet.get("version", 0) if wallet else 0
   253→   253→
   254→   254→    challenge = otp_service.create_challenge(tx_id)
   255→   255→
   256→   256→    now = _utcnow()
   257→   257→    doc = {
   258→   258→        "_id": tx_id,
   259→   259→        "idempotencyKey": idempotency_key,
   260→   260→        "agentId": agent_id,
   261→   261→        "orderId": order_oid,
   262→   262→        "amountPaise": amount_paise,
   263→   263→        "type": "otp_gated",
   264→   264→        "status": "awaiting_otp",
   265→   265→        "walletVersionBeforeTx": version_before,
   266→   266→        "razorpayOrderId": None,
   267→   267→        "razorpayPaymentId": None,
   268→   268→        "failureReason": None,
   269→   269→        "createdAt": now,
   270→   270→        "updatedAt": now,
   271→   271→        # Auto-clears stuck awaiting_otp locks (partial TTL index) — the janitor
   272→   272→        # ALSO marks them expired + cancels the order first, so records survive.
   273→   273→        "expiresAt": now + timedelta(seconds=config.OTP_TTL_SECONDS + 60),
   274→   274→    }
   275→   275→
   276→   276→    try:
   277→   277→        db.transactions().insert_one(doc)
   278→   278→    except Exception as err:  # noqa: BLE001
   279→   279→        if _is_duplicate_key(err):
   280→   280→            # Idempotency line 2: concurrent same-key execution raced past the
   281→   281→            # pre-check; the winner's stored snapshot is returned verbatim.
   282→   282→            winner = db.transactions().find_one({"idempotencyKey": idempotency_key})
   283→   283→            if winner is not None:
   284→   284→                return _response_for(winner)
   285→   285→        raise
   286→   286→
   287→   287→    response = {
   288→   288→        "status": "awaiting_otp",
   289→   289→        "transactionId": str(tx_id),
   290→   290→        "idempotencyKey": idempotency_key,
   291→   291→        "amountPaise": amount_paise,
   292→   292→        "orderNumber": order_number,
   293→   293→        "sessionId": session_id,
   294→   294→        "limitPaise": config.SPEND_LIMIT_PAISE,
   295→   295→    }
   296→   296→    if config.DEV_MODE:
   297→   297→        # DEV ONLY: in production this is sent via SMS/push, never returned in
   298→   298→        # the API response. otp_service stores ONLY the bcrypt hash.
   299→   299→        response["devOtp"] = challenge["otp"]
   300→   300→
   301→   301→    db.transactions().update_one(
   302→   302→        {"_id": tx_id},
   303→   303→        {"$set": {"resultSnapshot": dict(response), "otpChallengeId": None, "updatedAt": _utcnow()}},
   304→   304→    )
   305→   305→
   306→   306→    log_step(
   307→   307→        session_id, agent_id, "GUARDRAIL_OTP_REQUIRED",
   308→   308→        {
   309→   309→            "amountPaise": amount_paise,
   310→   310→            "limitPaise": config.SPEND_LIMIT_PAISE,
   311→   311→            "orderNumber": order_number,
   312→   312→            "transactionId": str(tx_id),
   313→   313→            "source": source,
   314→   314→            "message": "Amount exceeds delegated limit — human OTP verification required before any wallet debit.",
   315→   315→        },
   316→   316→        order_id=order_oid,
   317→   317→    )
   318→   318→    # Prompt the human gate in the UI (OTPModal.tsx listens for this).
   319→   319→    otp_payload = {
   320→   320→        "sessionId": session_id,
   321→   321→        "transactionId": str(tx_id),
   322→   322→        "amountPaise": amount_paise,
   323→   323→        "orderNumber": order_number,
   324→   324→    }
   325→   325→    if config.DEV_MODE:
   326→   326→        otp_payload["devOtp"] = challenge["otp"]
   327→   327→    ws_client.emit_to_room(session_id, "otp:required", otp_payload)
   328→   328→
   329→   329→    return dict(response)
   330→   330→
   331→   331→
   332→   332→# ---------------------------------------------------------------------------
   333→   333→# The ACID payment block — used by BOTH the autonomous path and the
   334→   334→# post-OTP-verified path. This is the only place that debits wallets.
   335→   335→# ---------------------------------------------------------------------------
   336→   336→
   337→   337→def _execute_payment(
   338→   338→    agent_id: str,
   339→   339→    session_id: str,
   340→   340→    idempotency_key: str,
   341→   341→    order_number: str,
   342→   342→    order_oid: ObjectId | None,
   343→   343→    amount_paise: int,
   344→   344→    tx_type: str,
   345→   345→    authorization: OtpAuthorization | None,
   346→   346→    source: str,
   347→   347→    pre_authorized: bool,
   348→   348→    existing_tx: dict | None = None,
   349→   349→) -> dict:
   350→   350→    tx_id = existing_tx["_id"] if existing_tx else ObjectId()
   351→   351→
   352→   352→    # Pre-flight wallet sanity (deterministic failures -> recorded, not silent).
   353→   353→    wallet = wallet_service.get_wallet(agent_id)
   354→   354→    if wallet is None:
   355→   355→        return _record_failed_transaction(
   356→   356→            agent_id, session_id, idempotency_key, order_oid, amount_paise, [],
   357→   357→            "WALLET_NOT_FOUND", source, tx_id=tx_id, order_number=order_number,
   358→   358→        )
   359→   359→    if wallet.get("status") != "active":
   360→   360→        return _record_failed_transaction(
   361→   361→            agent_id, session_id, idempotency_key, order_oid, amount_paise, [],
   362→   362→            "WALLET_FROZEN", source, tx_id=tx_id, order_number=order_number,
   363→   363→        )
   364→   364→    if wallet.get("balancePaise", 0) < amount_paise:
   365→   365→        return _record_failed_transaction(
   366→   366→            agent_id, session_id, idempotency_key, order_oid, amount_paise, [],
   367→   367→            "INSUFFICIENT_FUNDS", source, tx_id=tx_id, order_number=order_number,
   368→   368→        )
   369→   369→
   370→   370→    version_before = wallet.get("version", 0)
   371→   371→
   372→   372→    if not pre_authorized:
   373→   373→        log_step(
   374→   374→            session_id, agent_id, "GUARDRAIL_PASS",
   375→   375→            {
   376→   376→                "amountPaise": amount_paise,
   377→   377→                "limitPaise": config.SPEND_LIMIT_PAISE,
   378→   378→                "walletVersionBefore": version_before,
   379→   379→                "type": tx_type,
   380→   380→                "source": source,
   381→   381→            },
   382→   382→            order_id=order_oid,
   383→   383→        )
   384→   384→
   385→   385→    # ---- Multi-document ACID transaction -----------------------------------
   386→   386→    # debit + transaction-doc insert + Razorpay TEST order creation commit or
   387→   387→    # roll back TOGETHER: a Razorpay failure after the debit cannot leave the
   388→   388→    # wallet decremented with no corresponding record.
   389→   389→    # PRODUCTION NOTE: calling an external API inside the txn snapshot window is
   390→   390→    # acceptable here ONLY because this is TEST mode with no queue infra. The
   391→   391→    # production seam is a transactional outbox + worker (see razorpay_client.py).
   392→   392→    client = db.get_client()
   393→   393→    last_error: Exception | None = None
   394→   394→
   395→   395→    for attempt in range(1, 4):  # whole-txn retries on OCC exhaustion / transient races
   396→   396→        try:
   397→   397→            with client.start_session() as s:
   398→   398→                def _txn_callback(s):
   399→   399→                    # Guardrail is re-derived FRESH inside every attempt by
   400→   400→                    # wallet_service.execute_debit (no cached decision, no
   401→   401→                    # caller-supplied flag).
   402→   402→                    wallet_after = wallet_service.execute_debit(
   403→   403→                        agent_id, amount_paise, session=s, authorization=authorization
   404→   404→                    )
   405→   405→
   406→   406→                    now = _utcnow()
   407→   407→                    tx_doc = existing_tx or {
   408→   408→                        "_id": tx_id,
   409→   409→                        "idempotencyKey": idempotency_key,
   410→   410→                        "agentId": agent_id,
   411→   411→                        "orderId": order_oid,
   412→   412→                        "amountPaise": amount_paise,
   413→   413→                        "type": tx_type,
   414→   414→                        "status": "pending",
   415→   415→                        "walletVersionBeforeTx": version_before,
   416→   416→                        "razorpayOrderId": None,
   417→   417→                        "razorpayPaymentId": None,
   418→   418→                        "failureReason": None,
   419→   419→                        "createdAt": now,
   420→   420→                        "updatedAt": now,
   421→   421→                        # Generous pending TTL (stuck-lock backstop; the janitor
   422→   422→                        # expires orders/records before this ever fires).
   423→   423→                        "expiresAt": now + timedelta(seconds=config.RECOVERY_SESSION_TTL_SECONDS + 60),
   424→   424→                    }
   425→   425→                    if not existing_tx:
   426→   426→                        # Idempotency line 2 (inside txn): unique index races resolve
   427→   427→                        # to the winner's snapshot.
   428→   428→                        try:
   429→   429→                            db.transactions().insert_one(tx_doc, session=s)
   430→   430→                        except Exception as err:  # noqa: BLE001
   431→   431→                            if _is_duplicate_key(err):
   432→   432→                                winner = db.transactions().find_one(
   433→   433→                                    {"idempotencyKey": idempotency_key}, session=s
   434→   434→                                )
   435→   435→                                if winner is not None:
   436→   436→                                    return ("duplicate", winner, None)
   437→   437→                            raise
   438→   438→                    else:
   439→   439→                        db.transactions().update_one(
   440→   440→                            {"_id": tx_id, "status": {"$in": ["awaiting_otp", "otp_verified"]}},
   441→   441→                            {"$set": {"status": "pending", "updatedAt": now}},
   442→   442→                            session=s,
   443→   443→                        )
   444→   444→
   445→   445→                    # Razorpay TEST order (REAL API call — failures abort the txn
   446→   446→                    # and roll the debit back).
   447→   447→                    rzp_order = razorpay_client.create_test_order(
   448→   448→                        amount_paise,
   449→   449→                        receipt=order_number,
   450→   450→                        notes={
   451→   451→                            "orderNumber": order_number,
   452→   452→                            "agentId": agent_id,
   453→   453→                            "sessionId": session_id,
   454→   454→                            "type": tx_type,
   455→   455→                        },
   456→   456→                    )
   457→   457→                    rzp_order_id = rzp_order.get("id", "")
   458→   458→
   459→   459→                    db.transactions().update_one(
   460→   460→                        {"_id": tx_id},
   461→   461→                        {"$set": {"razorpayOrderId": rzp_order_id, "updatedAt": _utcnow()}},
   462→   462→                        session=s,
   463→   463→                    )
   464→   464→                    db.orders().update_one(
   465→   465→                        {"orderNumber": order_number},
   466→   466→                        {
   467→   467→                            "$set": {
   468→   468→                                "razorpayOrderId": rzp_order_id,
   469→   469→                                "status": "payment_pending",
   470→   470→                                "updatedAt": _utcnow(),
   471→   471→                            }
   472→   472→                        },
   473→   473→                        session=s,
   474→   474→                    )
   475→   475→                    return ("ok", wallet_after, rzp_order_id)
   476→   476→
   477→   477→                outcome = s.with_transaction(_txn_callback)
   478→   478→                kind, wallet_after, rzp_order_id = outcome[0], outcome[1], outcome[2]
   479→   479→
   480→   480→                if kind == "duplicate":
   481→   481→                    return _response_for(wallet_after)  # winner's snapshot, verbatim
   482→   482→
   483→   483→            response = {
   484→   484→                "status": "pending",  # awaiting Razorpay payment.captured webhook
   485→   485→                "transactionId": str(tx_id),
   486→   486→                "idempotencyKey": idempotency_key,
   487→   487→                "amountPaise": amount_paise,
   488→   488→                "orderNumber": order_number,
   489→   489→                "razorpayOrderId": rzp_order_id,
   490→   490→                "walletBalancePaise": wallet_after.get("balancePaise"),
   491→   491→                "message": "Payment initiated — order awaiting Razorpay confirmation.",
   492→   492→            }
   493→   493→            db.transactions().update_one(
   494→   494→                {"_id": tx_id},
   495→   495→                {"$set": {"resultSnapshot": response, "updatedAt": _utcnow()}},
   496→   496→            )
   497→   497→
   498→   498→            log_step(
   499→   499→                session_id, agent_id, "ORDER_GENERATED",
   500→   500→                {
   501→   501→                    "orderNumber": order_number,
   502→   502→                    "razorpayOrderId": rzp_order_id,
   503→   503→                    "amountPaise": amount_paise,
   504→   504→                    "walletBalancePaise": wallet_after.get("balancePaise"),
   505→   505→                    "type": tx_type,
   506→   506→                    "source": source,
   507→   507→                },
   508→   508→                order_id=order_oid,
   509→   509→            )
   510→   510→            return dict(response)
   511→   511→
   512→   512→        except razorpay_client.RazorpayNotConfigured as err:
   513→   513→            # Txn aborted: wallet debit already rolled back with the insert.
   514→   514→            return _record_failed_transaction(
   515→   515→                agent_id, session_id, idempotency_key, order_oid, amount_paise, [],
   516→   516→                "RAZORPAY_NOT_CONFIGURED", source, tx_id=tx_id, order_number=order_number,
   517→   517→                cancel_order=True,
   518→   518→            )
   519→   519→        except razorpay_client.RazorpayCallFailed as err:
   520→   520→            return _record_failed_transaction(
   521→   521→                agent_id, session_id, idempotency_key, order_oid, amount_paise, [],
   522→   522→                f"RAZORPAY_ORDER_FAILED: {err}", source, tx_id=tx_id,
   523→   523→                order_number=order_number, cancel_order=True,
   524→   524→            )
   525→   525→        except wallet_service.GuardrailViolation as err:
   526→   526→            # Someone attempted a >limit debit without verified OTP authorization
   527→   527→            # (defense-in-depth; execute_transaction routes those to the OTP path
   528→   528→            # first). Fail CLOSED, record the attempt.
   529→   529→            return _record_failed_transaction(
   530→   530→                agent_id, session_id, idempotency_key, order_oid, amount_paise, [],
   531→   531→                f"GUARDRAIL_VIOLATION: {err.code}", source, tx_id=tx_id,
   532→   532→                order_number=order_number, cancel_order=True,
   533→   533→            )
   534→   534→        except wallet_service.ConcurrentModificationMaxRetries as err:
   535→   535→            # Subclass of WalletError — MUST precede the generic WalletError
   536→   536→            # handler so the retry branch is actually reachable.
   537→   537→            last_error = err
   538→   538→            continue  # retry the whole ACID txn with a fresh snapshot
   539→   539→        except wallet_service.WalletError as err:
   540→   540→            return _record_failed_transaction(
   541→   541→                agent_id, session_id, idempotency_key, order_oid, amount_paise, [],
   542→   542→                err.code, source, tx_id=tx_id, order_number=order_number,
   543→   543→                cancel_order=True,
   544→   544→            )
   545→   545→
   546→   546→    return _record_failed_transaction(
   547→   547→        agent_id, session_id, idempotency_key, order_oid, amount_paise, [],
   548→   548→        f"CONCURRENT_MODIFICATION_MAX_RETRIES: {last_error}", source,
   549→   549→        tx_id=tx_id, order_number=order_number, cancel_order=True,
   550→   550→    )

   553→   553→def _is_duplicate_key(err: Exception) -> bool:
   554→   554→    code = getattr(err, "code", None)
   555→   555→    if code == 11000:
   556→   556→        return True
   557→   557→    msg = str(err)
   558→   558→    return "E11000" in msg or "duplicate key" in msg.lower()

   561→   561→def _record_failed_transaction(
   562→   562→    agent_id: str,
   563→   563→    session_id: str,
   564→   564→    idempotency_key: str,
   565→   565→    order_oid: ObjectId | None,
   566→   566→    amount_paise: int,
   567→   567→    items: list,
   568→   568→    reason: str,
   569→   569→    source: str,
   570→   570→    tx_id: ObjectId | None = None,
   571→   571→    order_number: str | None = None,
   572→   572→    cancel_order: bool = False,
   573→   573→) -> dict:
   574→   574→    """Record a deterministic failure so idempotent replays return it verbatim."""
   575→   575→    tx_id = tx_id or ObjectId()
   576→   576→    now = _utcnow()
   577→   577→    response = {
   578→   578→        "status": "failed",
   579→   579→        "transactionId": str(tx_id),
   580→   580→        "idempotencyKey": idempotency_key,
   581→   581→        "amountPaise": amount_paise,
   582→   582→        "orderNumber": order_number,
   583→   583→        "failureReason": reason,
   584→   584→    }
   585→   585→    try:
   586→   586→        db.transactions().update_one(
   587→   587→            {"_id": tx_id},
   588→   588→            {
   589→   589→                "$set": {
   590→   590→                    "idempotencyKey": idempotency_key,
   591→   591→                    "agentId": agent_id,
   592→   592→                    "orderId": order_oid,
   593→   593→                    "amountPaise": amount_paise,
   594→   594→                    "type": "autonomous",
   595→   595→                    "status": "failed",
   596→   596→                    "walletVersionBeforeTx": (wallet_service.get_wallet(agent_id) or {}).get("version", 0),
   597→   597→                    "razorpayOrderId": None,
   598→   598→                    "razorpayPaymentId": None,
   599→   599→                    "failureReason": reason,
   600→   600→                    "createdAt": now,
   601→   601→                    "updatedAt": now,
   602→   602→                    "resultSnapshot": response,
   603→   603→                }
   604→   604→            },
   605→   605→            upsert=True,
   606→   606→        )
   607→   607→    except Exception as err:  # noqa: BLE001
   608→   608→        if not _is_duplicate_key(err):
   609→   609→            raise
   610→   610→
   611→   611→    if cancel_order and order_number:
   612→   612→        cancel_order_and_release_stock(order_number, reason, agent_id, session_id)
   613→   613→
   614→   614→    log_step(
   615→   615→        session_id, agent_id, "ORDER_CANCELLED",
   616→   616→        {"orderNumber": order_number, "reason": reason, "source": source},
   617→   617→        order_id=order_oid,
   618→   618→    )
   619→   619→    return dict(response)
   620→   620→

   626→   626→def verify_otp(transaction_id: str, otp: str, session_id_hint: str = "") -> tuple[dict, int]:
   627→   627→    """Verify the human-entered OTP and, on success, execute the OTP-gated payment.
   628→   628→
   629→   629→    Returns (response, http_status).
   630→   630→    """
   631→   631→    try:
   632→   632→        tx_oid = ObjectId(transaction_id)
   633→   633→    except (InvalidId, TypeError):
   634→   634→        raise TransactionError("INVALID_TRANSACTION_ID", "transactionId must be a Mongo ObjectId hex.")
   635→   635→
   636→   636→    tx = db.transactions().find_one({"_id": tx_oid})
   637→   637→    if tx is None:
   638→   638→        raise TransactionError("TRANSACTION_NOT_FOUND", f"No transaction {transaction_id}.", 404)
   639→   639→
   640→   640→    if tx.get("status") != "awaiting_otp":
   641→   641→        # Idempotent replay: already verified / completed / failed / rejected.
   642→   642→        return _response_for(tx), 200
   643→   643→
   644→   644→    agent_id = tx["agentId"]
   645→   645→    session_id = tx.get("resultSnapshot", {}).get("sessionId") or session_id_hint or f"tx-{transaction_id}"
   646→   646→    # Prefer the sessionId captured at creation for room targeting.
   647→   647→    order_doc = db.orders().find_one({"_id": tx.get("orderId")}) if tx.get("orderId") else None
   648→   648→    order_number = order_doc.get("orderNumber") if order_doc else None
   649→   649→
   650→   650→    outcome, challenge = otp_service.verify_and_claim(tx_oid, otp or "")
   651→   651→
   652→   652→    if outcome == "verified":
   653→   653→        log_step(
   654→   654→            session_id, agent_id, "OTP_VERIFIED",
   655→   655→            {"transactionId": str(tx_oid), "attempts": challenge.get("attempts"), "orderNumber": order_number},
   656→   656→            order_id=tx.get("orderId"),
   657→   657→        )
   658→   658→        db.transactions().update_one(
   659→   659→            {"_id": tx_oid, "status": "awaiting_otp"},
   660→   660→            {"$set": {"status": "otp_verified", "updatedAt": _utcnow()}},
   661→   661→        )
   662→   662→        tx["status"] = "otp_verified"
   663→   663→
   664→   664→        # Server-side proof — the ONLY way an above-limit debit can proceed.
   665→   665→        authorization = OtpAuthorization(agent_id, tx_oid, tx["amountPaise"])
   666→   666→        response = _execute_payment(
   667→   667→            agent_id, session_id, tx["idempotencyKey"], order_number or "",
   668→   668→            tx.get("orderId"), tx["amountPaise"], tx_type="otp_gated",
   669→   669→            authorization=authorization, source="otp_verify",
   670→   670→            pre_authorized=True, existing_tx=tx,
   671→   671→        )
   672→   672→        return response, 200
   673→   673→
   674→   674→    if outcome == "already_verified":
   675→   675→        return _response_for(tx), 200
   676→   676→
   677→   677→    if outcome == "max_attempts":
   678→   678→        db.transactions().update_one(
   679→   679→            {"_id": tx_oid, "status": "awaiting_otp"},
   680→   680→            {"$set": {"status": "rejected", "failureReason": "OTP_MAX_ATTEMPTS", "updatedAt": _utcnow()}},
   681→   681→        )
   682→   682→        db.transactions().update_one({"_id": tx_oid}, {"$unset": {"expiresAt": ""}})
   683→   683→        if order_number:
   684→   684→            cancel_order_and_release_stock(order_number, "OTP_MAX_ATTEMPTS", agent_id, session_id)
   685→   685→        else:
   686→   686→            log_step(session_id, agent_id, "ORDER_CANCELLED",
   687→   687→                     {"transactionId": str(tx_oid), "reason": "OTP_MAX_ATTEMPTS"})
   688→   688→        return (
   689→   689→            {"status": "rejected", "transactionId": str(tx_oid),
   690→   690→             "message": "3 failed OTP attempts — transaction rejected and stock released."},
   691→   691→            401,
   692→   692→        )
   693→   693→
   694→   694→    if outcome == "expired":
   695→   695→        db.transactions().update_one(
   696→   696→            {"_id": tx_oid, "status": "awaiting_otp"},
   697→   697→            {"$set": {"status": "expired", "failureReason": "OTP_EXPIRED", "updatedAt": _utcnow()},
   698→   698→             "$unset": {"expiresAt": ""}},
   699→   699→        )
   700→   700→        if order_number:
   701→   701→            cancel_order_and_release_stock(order_number, "OTP_EXPIRED", agent_id, session_id)
   702→   702→        return (
   703→   703→            {"status": "expired", "transactionId": str(tx_oid), "message": "OTP challenge expired (5-minute TTL)."},
   704→   704→            410,
   705→   705→        )
   706→   706→
   707→   707→    # outcome == 'wrong'
   708→   708→    left = otp_service.attempts_left(tx_oid)
   709→   709→    return (
   710→   710→        {"status": "awaiting_otp", "transactionId": str(tx_oid),
   711→   711→         "attemptsLeft": left, "message": "Incorrect OTP."},
   712→   712→        401,
   713→   713→    )
   714→   714→
   715→   715→
   716→   716→# ---------------------------------------------------------------------------
   717→   717→# Janitor — expires stuck awaiting_otp / pending transactions BEFORE the TTL
   718→   718→# index deletes them, cancelling the order and releasing reserved stock.
   719→   719→# ---------------------------------------------------------------------------
   720→   720→
   721→   721→def expire_stale_transactions(limit: int = 100) -> int:
   722→   722→    now = _utcnow()
   723→   723→    stale = db.transactions().find(
   724→   724→        {"status": {"$in": ["pending", "awaiting_otp"]}, "expiresAt": {"$lt": now}},
   725→   725→        {"_id": 1, "agentId": 1, "orderId": 1, "idempotencyKey": 1, "status": 1},
   726→   726→    ).limit(limit)
   727→   727→    expired = 0
   728→   728→    for tx in stale:
   729→   729→        order_doc = db.orders().find_one({"_id": tx.get("orderId")}) if tx.get("orderId") else None
   730→   730→        # PRODUCTION NOTE: expiring a debit-ed 'pending' transaction would also
   731→   731→        # trigger a wallet credit (refund) here; out of scope for the TEST-mode build.
   732→   732→        db.transactions().update_one(
   733→   733→            {"_id": tx["_id"], "status": {"$in": ["pending", "awaiting_otp"]}},
   734→   734→            {"$set": {"status": "expired", "failureReason": "TTL_EXPIRED", "updatedAt": now},
   735→   735→             "$unset": {"expiresAt": ""}},
   736→   736→        )
   737→   737→        if order_doc and order_doc.get("orderNumber"):
   738→   738→            cancel_order_and_release_stock(
   739→   739→                order_doc["orderNumber"], "TTL_EXPIRED", tx["agentId"], f"tx-{tx['_id']}"
   740→   740→            )
   741→   741→        expired += 1
   742→   742→    return expired
